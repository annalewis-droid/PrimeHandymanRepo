2. Frontend Implementation: Vanilla JavaScript Controller
This script initializes the Supabase client, intercepts the form submission, manages the multi-state UI (Submitting, Success, Failure), and structures the exact payload required by your contact_submissions contract.
3. Architectural ReviewZero Frameworks: Completely devoid of build steps. Highly cacheable on Cloudflare.Contract Alignment: The contact_submissions table interface receives exactly the expected data types and properties, honoring the mandate to concatenate the user's name into customer_name.  Error Handling: Implements rigid two-tier validation with clear feedback cycles preventing duplicate button clicks and gracefully catching network/database failures.
	// Initialize Supabase Client
// Note: These keys will be visible in the source code on Cloudflare Pages.
const SUPABASE_URL = 'YOUR_SUPABASE_PROJECT_URL';
const SUPABASE_ANON_KEY = 'YOUR_SUPABASE_ANON_KEY';
const supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

document.addEventListener('DOMContentLoaded', () => {
  const form = document.getElementById('prime-contact-form');
  const submitBtn = document.getElementById('submit-btn');
  const successMsg = document.getElementById('form-success');
  const errorMsg = document.getElementById('form-error');

  form.addEventListener('submit', async (e) => {
    e.preventDefault();

    // 1. Validate Turnstile (Anti-Spam Guardrail)
    const formData = new FormData(form);
    const turnstileToken = formData.get('cf-turnstile-response');
    
    if (!turnstileToken) {
        showError('Please complete the anti-spam verification.');
        return;
    }

    // 2. Set Submitting State
    setLoadingState(true);
    errorMsg.style.display = 'none';
    successMsg.style.display = 'none';

    // 3. Construct Data Contract Payload
    // Concatenating First and Last name to satisfy the singular customer_name string property priority
    const firstName = formData.get('firstName').trim();
    const lastName = formData.get('lastName').trim();
    const customerName = `${firstName} ${lastName}`;

    const payload = {
      brand_slug: 'prime-handyman', // Mandatory default for multi-tenant tracking
      customer_name: customerName,
      email: formData.get('email').trim(),
      phone: formData.get('phone').trim() || null,
      message: formData.get('message').trim()
    };

    // 4. Asynchronous Database Insertion
    try {
      const { error } = await supabase
        .from('contact_submissions')
        .insert([payload]);

      if (error) throw error;

      // 5. Set Success State
      form.reset();
      turnstile.reset(); // Reset the Turnstile widget
      successMsg.style.display = 'block';

    } catch (error) {
      // 6. Set Failure State
      console.error('Supabase Transaction Failure:', error);
      showError('Our system encountered an issue processing your request. Please try again later.');
    } finally {
      // Revert UI from Submitting State
      setLoadingState(false);
    }
  });

  // UI State Handlers
  function setLoadingState(isLoading) {
    if (isLoading) {
      submitBtn.disabled = true;
      submitBtn.textContent = 'Submitting...';
      // Disable inputs to prevent duplicate submission mutations
      Array.from(form.elements).forEach(el => el.disabled = true);
    } else {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Submit Inquiry';
      Array.from(form.elements).forEach(el => el.disabled = false);
    }
  }

  function showError(message) {
    errorMsg.textContent = message;
    errorMsg.style.display = 'block';
  }
});
